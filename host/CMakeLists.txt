cmake_minimum_required(VERSION 3.18)
project(freertos_lab_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)
if(NOT WIN32)
  message(STATUS "Windows COM bridge executables are Windows-only; portable protocol tests remain available")
else()
  add_executable(monitor ../tools/monitor.c ../tools/protocol.c)
  add_executable(cli ../tools/cli.c ../tools/protocol.c)
  target_link_libraries(monitor PRIVATE ws2_32)
  target_link_libraries(cli PRIVATE ws2_32)
endif()
enable_testing()
add_executable(host_tests ../tests/test_host.c ../tools/protocol.c)
add_test(NAME host_protocol COMMAND host_tests)
add_executable(timing_tests ../tests/test_timing.c ../tools/timing.c)
add_test(NAME host_timing_10s COMMAND timing_tests)
set_tests_properties(host_timing_10s PROPERTIES TIMEOUT 20)
if(WIN32)
  target_sources(monitor PRIVATE ../tools/timing.c)
endif()
if(WIN32)
  add_executable(bridge_tests ../tests/test_bridge_win.c ../tools/protocol.c ../tools/timing.c)
  target_link_libraries(bridge_tests PRIVATE ws2_32)
  add_test(NAME windows_mock_bridge COMMAND bridge_tests)
endif()
add_executable(firmware_tests ../tests/test_firmware.c ../tools/protocol.c)
target_include_directories(firmware_tests PRIVATE ../tests/fakes ../Core ../App)
add_test(NAME firmware_dispatch_doubles COMMAND firmware_tests)
if(WIN32)
  add_executable(cli_tests ../tests/test_cli_win.c ../tools/protocol.c)
  target_link_libraries(cli_tests PRIVATE ws2_32)
  add_test(NAME windows_cli_help COMMAND cli_tests)
endif()
